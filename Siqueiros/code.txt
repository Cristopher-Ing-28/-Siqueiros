        IndiceHistorial = 0    % Posición actual en el historial

    methods (Access = private)
           % Guarda el estado actual en el historial
        function guardarEstado(app, nuevaImagen)
            % Elimina estados de rehacer si nos encontramos en medio del historial
            if app.IndiceHistorial < length(app.HistorialImagenes)
                app.HistorialImagenes = app.HistorialImagenes(1:app.IndiceHistorial);
            end
            
            % Agregar el nuevo estado y actualizar el índice
            app.IndiceHistorial = app.IndiceHistorial + 1;
            app.HistorialImagenes{app.IndiceHistorial} = nuevaImagen;
            app.ImagenProcesada = nuevaImagen;
        end
        
        % Actualiza los visores UIAxes_2 y UIAxes_4
        function actualizarInterfazProcesada(app)
            img = app.ImagenProcesada;
            
            % 1. Renderizar Imagen Procesada
            imshow(img, 'Parent', app.UIAxes_2);
            
            % 2. Renderizar Histograma Restaurado en UIAxes_4
            if size(img, 3) == 3
                grisHist = rgb2gray(img);
            else
                grisHist = img;
            end
            
            [counts, binLocations] = imhist(grisHist);
            
            cla(app.UIAxes_4);
            bar(app.UIAxes_4, binLocations, counts, 'BarWidth', 1, 'FaceColor', [0 0.4470 0.7410], 'EdgeColor', 'none');
            app.UIAxes_4.XLim = [0 255];
            app.UIAxes_4.YLimMode = 'auto';
            app.UIAxes_4.YColor = [0.9 0.9 0.9];
            app.UIAxes_4.XColor = [0.9 0.9 0.9];
            
            % 3. Garantizar Histograma Original en UIAxes_3
            if ~isempty(app.ImagenOriginal)
                if size(app.ImagenOriginal, 3) == 3
                    grisOrig = rgb2gray(app.ImagenOriginal);
                else
                    grisOrig = app.ImagenOriginal;
                end
                [countsOrig, binLocOrig] = imhist(grisOrig);
                cla(app.UIAxes_3);
                bar(app.UIAxes_3, binLocOrig, countsOrig, 'BarWidth', 1, 'FaceColor', [0 0.4470 0.7410], 'EdgeColor', 'none');
                app.UIAxes_3.XLim = [0 255];
                app.UIAxes_3.YLimMode = 'auto';
                app.UIAxes_3.YColor = [0.9 0.9 0.9];
                app.UIAxes_3.XColor = [0.9 0.9 0.9];
            end
        end
    end

        % Button pushed function: ImportarButton
        function ImportarButtonPushed(app, event)
            fileFilters = {
                '*.png;*.jpg;*.jpeg;*.tif;*.tiff;*.bmp', 'Archivos de imagen (*.png, *.jpg, *.jpeg, *.tif, *.bmp)'; ...
                '*.png',  'Imágenes PNG (*.png)'; ...
                '*.jpg;*.jpeg', 'Imágenes JPEG (*.jpg, *.jpeg)'; ...
                '*.tif;*.tiff', 'Imágenes TIFF (*.tif, *.tiff)'; ...
                '*.bmp',  'Imágenes Bitmap (*.bmp)'; ...
                '*.*',    'Todos los archivos (*.*)'};
            [fileName, folderPath] = uigetfile(fileFilters, 'Seleccionar imagen antigua');
            if isequal(fileName, 0) || isequal(folderPath, 0)
                return; % El usuario canceló la selección
            end
            try
                fullImagePath = fullfile(folderPath, fileName);
                img = imread(fullImagePath);
    
                app.ImagenOriginal = img;
                app.ImagenProcesada = img;
    
                % Resetear historial
                app.HistorialImagenes = {img};
                app.IndiceHistorial = 1;
    
                imshow(img, 'Parent', app.UIAxes);
                
                % Dibujar Histograma Original en UIAxes_3
                if size(img, 3) == 3
                    grisOrig = rgb2gray(img);
                else
                    grisOrig = img;
                end
                [countsOrig, binLocOrig] = imhist(grisOrig);
                cla(app.UIAxes_3);
                bar(app.UIAxes_3, binLocOrig, countsOrig, 'BarWidth', 1, 'FaceColor', [0 0.4470 0.7410], 'EdgeColor', 'none');
                app.UIAxes_3.XLim = [0 255];
                app.UIAxes_3.YLimMode = 'auto';
                app.UIAxes_3.YColor = [0.9 0.9 0.9];
                app.UIAxes_3.XColor = [0.9 0.9 0.9];
    
                % Actualizar visor del procesado con la misma imagen inicial
                app.actualizarInterfazProcesada();
    
                app.EstadodelaimagenLabel.Text = sprintf('Estado: Importación exitosa (%s)', fileName);
            catch ME
                app.EstadodelaimagenLabel.Text = 'Estado: Fallo en la importación de la imagen.';
            end
        end

        % Button pushed function: RayonesButton
        function RayonesButtonPushed(app, event)
            if isempty(app.ImagenProcesada)
                uialert(app.UIFigure, 'Por favor, importe una imagen primero.', 'Falta Imagen');
                return;
            end
            
            app.EstadodelaimagenLabel.Text = 'Estado: Eliminando rayones...';
            drawnow;
            
            % 1. Obtener imagen actual
            img = app.ImagenProcesada;
            
            % Convertir a escala de grises para la máscara
            if size(img, 3) == 3
                grayImg = rgb2gray(img);
            else
                grayImg = img;
            end
            
            % 2. Top-Hat morfológico con un elemento estructurante más amplio (para rayones gruesos)
            se = strel('disk', 6); % Aumentado de 4 a 6
            tophatImg = imtophat(grayImg, se);
            
            % Umbral adaptativo basado en la mediana y desviación estándar del Top-Hat
            umbral = median(tophatImg(:)) + 1.8 * std(double(tophatImg(:)));
            mascaraRayones = tophatImg > umbral;
            
            % Engrosar la máscara para cubrir bordes desgastados del rayón
            mascaraRayones = imdilate(mascaraRayones, strel('disk', 2));
            
            % 3. Aplicar Inpainting (regionfill) sobre la máscara generada
            if size(img, 3) == 3
                imgRestaurada = img;
                for c = 1:3
                    imgRestaurada(:,:,c) = regionfill(img(:,:,c), mascaraRayones);
                end
            else
                imgRestaurada = regionfill(img, mascaraRayones);
            end
            
            % 4. Actualizar variable en proceso
            app.ImagenProcesada = imgRestaurada;
            
            % 5. Mostrar imagen procesada
            imshow(imgRestaurada, 'Parent', app.UIAxes_2); 
            
            % 6. Renderizar Histograma Restaurado en UIAxes_4
            if size(imgRestaurada, 3) == 3
                grisHist = rgb2gray(imgRestaurada);
            else
                grisHist = imgRestaurada;
            end
            
            [counts, binLocations] = imhist(grisHist);
            bar(app.UIAxes_4, binLocations, counts, 'BarWidth', 1, 'FaceColor', [0.8 0.8 0.8], 'EdgeColor', 'none');
            app.UIAxes_4.XLim = [0 255];
            app.UIAxes_4.YColor = [0.9 0.9 0.9];
            app.UIAxes_4.XColor = [0.9 0.9 0.9];
            
            % 7. Renderizar Histograma Original en UIAxes_3 si aún está vacío
            if size(app.ImagenOriginal, 3) == 3
                grisOrig = rgb2gray(app.ImagenOriginal);
            else
                grisOrig = app.ImagenOriginal;
            end
            [countsOrig, binLocOrig] = imhist(grisOrig);
            bar(app.UIAxes_3, binLocOrig, countsOrig, 'BarWidth', 1, 'FaceColor', [0.8 0.8 0.8], 'EdgeColor', 'none');
            app.UIAxes_3.XLim = [0 255];
            app.UIAxes_3.YColor = [0.9 0.9 0.9];
            app.UIAxes_3.XColor = [0.9 0.9 0.9];

            app.guardarEstado(imgRestaurada);
            app.actualizarInterfazProcesada();
            app.EstadodelaimagenLabel.Text = 'Estado: Rayones eliminados correctamente.';
        end

        % Button pushed function: DegradacinButton
        function DegradacinButtonPushed(app, event)

           if isempty(app.ImagenOriginal) && isempty(app.ImagenProcesada)
            uialert(app.UIFigure, 'Por favor, importe una imagen primero.', 'Falta Imagen');
            return;
           end
        
            app.EstadodelaimagenLabel.Text = 'Estado: Restaurando imagen degradada por el sol...';
            drawnow;
        
            try
                if ~isempty(app.ImagenProcesada)
                    imgBase = app.ImagenProcesada;
                else
                    imgBase = app.ImagenOriginal;
                end
        
                % 1. Restauración HSV / CLAHE
                if size(imgBase, 3) == 3
                    hsvImg = rgb2hsv(imgBase);
                    hsvImg(:,:,2) = min(hsvImg(:,:,2) * 1.3, 1.0); 
                    hsvImg(:,:,3) = adapthisteq(hsvImg(:,:,3), 'ClipLimit', 0.02);
                    imgRestaurada = hsv2rgb(hsvImg);
                else
                    imgRestaurada = adapthisteq(imgBase, 'ClipLimit', 0.02);
                end
        
                % Filtro gaussiano suave
                imgRestaurada = imgaussfilt(imgRestaurada, 0.6);
                imgRestaurada = im2uint8(imgRestaurada);
        
                % 2. Actualizar propiedad en App Designer
                app.ImagenProcesada = imgRestaurada;
        
                % 3. Mostrar la imagen procesada
                imshow(imgRestaurada, 'Parent', app.UIAxes_2); 
        
                % 4. Renderizar Histograma Restaurado en UIAxes_4
                if size(imgRestaurada, 3) == 3
                    grisHist = rgb2gray(imgRestaurada);
                else
                    grisHist = imgRestaurada;
                end
                
                [counts, binLocations] = imhist(grisHist);
                bar(app.UIAxes_4, binLocations, counts, 'BarWidth', 1, 'FaceColor', [0 0.4470 0.7410], 'EdgeColor', 'none');
                app.UIAxes_4.XLim = [0 255];
                app.UIAxes_4.YColor = [0.9 0.9 0.9];
                app.UIAxes_4.XColor = [0.9 0.9 0.9];
                
                % 5. Renderizar Histograma Original en UIAxes_3 si aplica
                if size(app.ImagenOriginal, 3) == 3
                    grisOrig = rgb2gray(app.ImagenOriginal);
                else
                    grisOrig = app.ImagenOriginal;
                end
                [countsOrig, binLocOrig] = imhist(grisOrig);
                bar(app.UIAxes_3, binLocOrig, countsOrig, 'BarWidth', 1, 'FaceColor', [0 0.4470 0.7410], 'EdgeColor', 'none');
                app.UIAxes_3.XLim = [0 255];
                app.UIAxes_3.YColor = [0.9 0.9 0.9];
                app.UIAxes_3.XColor = [0.9 0.9 0.9];
                
                app.guardarEstado(imgRestaurada);
                app.actualizarInterfazProcesada();

            catch ME
                app.EstadodelaimagenLabel.Text = 'Estado: Error al aplicar la restauración por degradación.';
            end

        % Button pushed function: ReconstruccinButton
        function ReconstruccinButtonPushed(app, event)
            if isempty(app.ImagenProcesada)
                uialert(app.UIFigure, 'Por favor, importe una imagen primero.', 'Falta Imagen');
                return;
            end

            app.EstadodelaimagenLabel.Text = 'Estado: Reconstruyendo roturas y esquinas (M4)...';
            drawnow;

            try
                img = app.ImagenProcesada;

                if size(img, 3) == 3
                    grayImg = rgb2gray(img);
                else
                    grayImg = img;
                end

                % 1. Detección de fondo/papel faltante (zonas sobreexpuestas por escaneo > 225)
                mascaraVacios = (grayImg > 225);

                % 2. Limitar la reconstrucción de esquinas estrictamente al PERÍMETRO (25% exterior)
                % Esto aísla el centro de la imagen y protege ojos, nariz y boca
                [filas, columnas] = size(grayImg);
                mascaraPerimetro = false(filas, columnas);
                mF = round(filas * 0.25);
                mC = round(columnas * 0.25);
                mascaraPerimetro(1:mF, :) = true;
                mascaraPerimetro(end-mF+1:end, :) = true;
                mascaraPerimetro(:, 1:mC) = true;
                mascaraPerimetro(:, end-mC+1:end) = true;

                mascaraEsquinas = mascaraVacios & mascaraPerimetro;

                % 3. Detección de grietas morfológicas usando solo Top-Hat con alto umbral
                % Se remueve Bottom-Hat para evitar interpretar sombras oscuras del rostro como daños
                topHat = imtophat(grayImg, strel('disk', 4));
                umbralTopHat = median(topHat(:)) + 3.0 * std(double(topHat(:)));
                mascaraGrietas = topHat > umbralTopHat;

                % 4. Unir máscaras
                mascaraRoturas = mascaraEsquinas | mascaraGrietas;

                % 5. Limpieza y dilatación controlada
                mascaraRoturas = bwareaopen(mascaraRoturas, 30);
                mascaraRoturas = imdilate(mascaraRoturas, strel('disk', 3));

                if ~any(mascaraRoturas(:))
                    app.EstadodelaimagenLabel.Text = 'Estado: No se detectaron roturas en la imagen.';
                    return;
                end

                % 6. Inpainting canal por canal
                if size(img, 3) == 3
                    imgRestaurada = img;
                    for c = 1:3
                        imgRestaurada(:,:,c) = regionfill(img(:,:,c), mascaraRoturas);
                    end
                else
                    imgRestaurada = regionfill(img, mascaraRoturas);
                end

                app.guardarEstado(imgRestaurada);
                app.actualizarInterfazProcesada();
                app.EstadodelaimagenLabel.Text = 'Estado: Reconstrucción M4 completada exitosamente.';

            catch ME
                uialert(app.UIFigure, ME.message, 'Error en Reconstrucción M4');
                app.EstadodelaimagenLabel.Text = 'Estado: Error al ejecutar el módulo M4.';
            end
        end

        % Button pushed function: AdicindeColorButton
        function AdicindeColorButtonPushed(app, event)
            if isempty(app.ImagenProcesada)
                uialert(app.UIFigure, 'Por favor, importe una imagen primero.', 'Falta Imagen');
                return;
            end

            app.EstadodelaimagenLabel.Text = 'Estado: Colorizando por transferencia de luminancia L*a*b* (M6)...';
            drawnow;

            try
                img = app.ImagenProcesada;

                % 1. Extraer escala de grises / luminancia
                if size(img, 3) == 3
                    grayImg = rgb2gray(img);
                else
                    grayImg = img;
                end

                lum = im2double(grayImg);
                L = lum * 100; % Escala L* de 0 a 100

                % 2. Puntos de control para la curva de transferencia cromática (PDI determinista)
                % L_puntos: Niveles de brillo (0=Negro absoluto, 100=Blanco puro)
                % a_puntos: Verde (-) a Rojo (+)
                % b_puntos: Azul (-) a Amarillo (+)
                L_puntos = [0,  12,  30,  55,  75,  92, 100];
                a_puntos = [1,   2,   7,  13,   9,   2,   0];
                b_puntos = [2,   3,  10,  16,  11,   3,   1];

                % 3. Interpolación suave continua (pchip) para cada píxel de la matriz
                % Aplica el tono a TODAS las zonas según su brillo, sin máscaras circulares rígidas
                a = interp1(L_puntos, a_puntos, L, 'pchip');
                b = interp1(L_puntos, b_puntos, L, 'pchip');

                % 4. Filtrado suave morfológico para homogeneizar la transición cromática
                a = imgaussfilt(a, 1.5);
                b = imgaussfilt(b, 1.5);

                % 5. Reconstrucción de L*a*b* a RGB
                imgLAB = cat(3, L, a, b);
                imgRestaurada = lab2rgb(imgLAB);
                imgRestaurada = im2uint8(imgRestaurada);

                % 6. Actualizar historial e interfaz
                app.guardarEstado(imgRestaurada);
                app.actualizarInterfazProcesada();
                app.EstadodelaimagenLabel.Text = 'Estado: Colorización M6 completada exitosamente.';

            catch ME
                uialert(app.UIFigure, ME.message, 'Error en Colorización M6');
                app.EstadodelaimagenLabel.Text = 'Estado: Error al ejecutar la colorización M6.';
            end
        end

        % Button pushed function: ManchasButton
        function ManchasButtonPushed(app, event)
if isempty(app.ImagenProcesada)
        uialert(app.UIFigure, 'Por favor, importe una imagen primero.', 'Falta Imagen');
        return;
    end

    app.EstadodelaimagenLabel.Text = 'Estado: Disminuyendo manchas (M3)...';
    drawnow;

    try
        % --- Parámetros ajustables (calibrados para gotas en el lente) ---
        kFuerte    = 3.0;    % sigmas para semillas (más bajo = más sensible)
        kDebil     = 1.8;    % sigmas para extender la mancha (histéresis)
        factorEsc  = 0.25;   % escala para estimar el fondo (rápido)
        ventanaEsc = 25;     % impar; a escala 0.25 equivale a ~100 px reales
        areaMin    = 30;     % px mínimos para considerar mancha
        areaMaxFr  = 0.010;  % fracción máxima de la imagen por región
        ratioMax   = 3;      % eje mayor/menor (mayor = línea/rayo, se protege)
        solidezMin = 0.5;    % formas muy irregulares se protegen
        radioDilat = 5;      % cubre el halo difuso de la gota
        radioInp   = 15;     % radio de inpaintCoherent
        mostrarMascara = false;  % true para depurar la máscara
        % -----------------------------------------------------------------

        % 0. Normalizar clase: elimina la dependencia de uint8 vs double
        img = im2uint8(app.ImagenProcesada);
        esGris = size(img, 3) == 1;
        if esGris
            img = repmat(img, 1, 1, 3);
        end

        % 1. Lab: luminancia + croma
        lab = rgb2lab(img);
        L = lab(:,:,1);  A = lab(:,:,2);  B = lab(:,:,3);

        % 2. Fondo local robusto, estimado a escala reducida
        fondo = @(X) imresize( ...
                     medfilt2(imresize(X, factorEsc, 'bilinear'), ...
                              [ventanaEsc ventanaEsc], 'symmetric'), ...
                     [size(X,1), size(X,2)], 'bilinear');

        bgL = fondo(L);  bgA = fondo(A);  bgB = fondo(B);

        dL = L - bgL;
        dC = hypot(A - bgA, B - bgB);

        % 3. Puntaje normalizado (MAD); solo oscurecimientos en luminancia
        sL = max(1.4826 * median(abs(dL(:) - median(dL(:)))), 1);
        sC = max(1.4826 * median(dC(:)), 1);
        puntaje = max(max(-dL, 0) / sL, dC / sC);

        % 4. Histéresis: semillas fuertes extendidas a toda la mancha
        mascara = imreconstruct(puntaje > kFuerte, puntaje > kDebil);

        % 5. Filtrado por tamaño
        mascara = bwareaopen(mascara, areaMin);
        mascara = mascara & ~bwareaopen(mascara, round(areaMaxFr * numel(mascara)));

        % 6. Filtrado por forma: proteger líneas, rayo, arcoíris y estructuras alargadas
        cc = bwconncomp(mascara);
        if cc.NumObjects > 0
            st = regionprops(cc, 'Solidity', 'MajorAxisLength', 'MinorAxisLength');
            ratio = [st.MajorAxisLength] ./ max([st.MinorAxisLength], 1);
            conservar = ([st.Solidity] > solidezMin) & (ratio < ratioMax);
            mascara = ismember(labelmatrix(cc), find(conservar));
        end

        % 7. Expandir para cubrir halos de la mancha
        mascara = imdilate(mascara, strel('disk', radioDilat));

        nManchas = bwconncomp(mascara).NumObjects;
        if nManchas == 0
            app.EstadodelaimagenLabel.Text = ...
                'Estado: No se detectaron manchas (prueba bajar kFuerte).';
            return;
        end

        % Depuración visual opcional
        if mostrarMascara
            figure; imshow(labeloverlay(img, mascara, 'Transparency', 0.5));
        end

        % 8. Reparación
        try
            imgRestaurada = inpaintCoherent(img, mascara, ...
                'Radius', radioInp, 'SmoothingFactor', 2);   % Requiere R2019a+ o Computer Vision Toolbox
        catch
            imgRestaurada = img;
            for c = 1:3
                imgRestaurada(:,:,c) = regionfill(img(:,:,c), mascara);
            end
        end

        if esGris
            imgRestaurada = imgRestaurada(:,:,1);
        end

        % 9. Guardar y actualizar interfaz
        app.guardarEstado(imgRestaurada);
        app.actualizarInterfazProcesada();
        app.EstadodelaimagenLabel.Text = sprintf( ...
            'Estado: %d manchas reparadas (%.2f%% de la imagen).', ...
            nManchas, 100 * nnz(mascara) / numel(mascara));

    catch ME
        uialert(app.UIFigure, ME.message, 'Error en Módulo 3');
        app.EstadodelaimagenLabel.Text = 'Estado: Error al aplicar la disminución de manchas.';
    end
        end
    end
